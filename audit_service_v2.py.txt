import json
from collections import defaultdict
from typing import Dict, List, Set, Tuple
import difflib
import csv
from pathlib import Path
from dataclasses import dataclass
from datetime import datetime

KB_FILE_PATH = "master_kb.json"
REPORT_DIR = Path("audit_reports")
REPORT_DIR.mkdir(exist_ok=True)

# 1. TASSONOMIA ATTESA (Regole di validazione)
EXPECTED_SERVICES_BY_TYPE = {
    "Ospedale Hub (Emergenza)": [
        "Pronto Soccorso Generale",
        "Terapia Intensiva",
        "Cardiologia"
    ],
    "Ospedale Spoke": [
        "Pronto Soccorso Generale",
        "Medicina Generale"
    ],
    "SerDP": [
        "Trattamento Dipendenze",
        "Assistenza Psicologica"
    ],
}

# 2. MAPPING DI NORMALIZZAZIONE (Sinonimi -> Forma canonica)
SERVICE_NORMALIZATION = {
    "Pronto soccorso": "Pronto Soccorso Generale",
    "Pronto Soccorso": "Pronto Soccorso Generale",
    "PS": "Pronto Soccorso Generale",
    "Shock Room": "Rianimazione e Stabilizzazione",
    "ICU": "Terapia Intensiva",
    "Terapia intensiva": "Terapia Intensiva",
}

@dataclass
class ServiceAnomalies:
    """Raccogli tutte le anomalie rilevate"""
    missing_required: List[Dict] = None
    empty_services: List[Dict] = None
    duplicates: List[Dict] = None
    outliers: List[Dict] = None  # Servizi rari
    
    def __post_init__(self):
        self.missing_required = self.missing_required or []
        self.empty_services = self.empty_services or []
        self.duplicates = self.duplicates or []
        self.outliers = self.outliers or []

def fuzzy_match_service(service: str, candidates: List[str], threshold: float = 0.8) -> Tuple[bool, str]:
    """
    Fuzzy matching robusto: se il servizio è simile a un candidato per almeno threshold,
    ritorna la forma canonica.
    
    Es: "Pronto Soccorso Generale" (inputato male) -> Match con "Pronto Soccorso Generale"
    """
    service_lower = service.lower().strip()
    for candidate in candidates:
        ratio = difflib.SequenceMatcher(None, service_lower, candidate.lower()).ratio()
        if ratio >= threshold:
            return True, candidate
    return False, None

def normalize_service(service: str) -> str:
    """
    Normalizza il servizio usando il mapping predefinito.
    Se non trova una corrispondenza, ritorna il servizio pulito.
    """
    service_clean = service.strip()
    
    # Controllo esatto (case-insensitive)
    for key, canonical in SERVICE_NORMALIZATION.items():
        if service_clean.lower() == key.lower():
            return canonical
    
    # Altrimenti ritorna come è (dopo pulizia)
    return service_clean

def load_kb(filepath: str) -> dict:
    with open(filepath, 'r', encoding='utf-8') as f:
        return json.load(f)

def audit_services(data: dict) -> Tuple[Dict, ServiceAnomalies]:
    """
    Esegue audit completo. Ritorna:
    - dict con statistiche generali
    - ServiceAnomalies con tutte le anomalie
    """
    facilities = data.get("facilities", [])
    
    all_services_frequency = defaultdict(int)
    services_by_type = defaultdict(set)
    anomalies = ServiceAnomalies()
    
    print(f"📋 Inizio audit su {len(facilities)} strutture...\n")
    
    for facility in facilities:
        fid = facility.get("id", "UNKNOWN_ID")
        ftype = facility.get("tipologia", "UNKNOWN_TYPE")
        services = facility.get("servizi_disponibili", [])
        
        # Anomalia 1: Servizi vuoti
        if not services:
            anomalies.empty_services.append({
                "id": fid,
                "tipologia": ftype,
                "errore": "Lista 'servizi_disponibili' vuota o mancante"
            })
            continue
        
        # Normalizza tutti i servizi
        normalized_services = []
        for service in services:
            normalized = normalize_service(service)
            normalized_services.append(normalized)
            all_services_frequency[normalized] += 1
            services_by_type[ftype].add(normalized)
        
        # Anomalia 2: Servizi richiesti per tipologia
        expected_services = EXPECTED_SERVICES_BY_TYPE.get(ftype, [])
        for expected in expected_services:
            # Fuzzy match robusto
            matched, canonical = fuzzy_match_service(expected, normalized_services, threshold=0.8)
            if not matched:
                anomalies.missing_required.append({
                    "id": fid,
                    "tipologia": ftype,
                    "servizio_atteso": expected,
                    "servizi_attuali": normalized_services
                })
        
        # Anomalia 3: Duplicati di servizio nella stessa struttura
        if len(normalized_services) != len(set(normalized_services)):
            counter = defaultdict(int)
            for svc in normalized_services:
                counter[svc] += 1
            dupes = {s: count for s, count in counter.items() if count > 1}
            if dupes:
                anomalies.duplicates.append({
                    "id": fid,
                    "tipologia": ftype,
                    "duplicati": dupes
                })
    
    # Anomalia 4: Outliers (servizi molto rari)
    threshold_outlier = max(1, len(facilities) // 50)  # Se offerto da <2% strutture
    for service, count in all_services_frequency.items():
        if count < threshold_outlier:
            anomalies.outliers.append({
                "servizio": service,
                "frequenza": count,
                "note": "Servizio offerto da meno del 2% delle strutture"
            })
    
    stats = {
        "total_facilities": len(facilities),
        "total_unique_services": len(all_services_frequency),
        "anomaly_count": {
            "empty": len(anomalies.empty_services),
            "missing_required": len(anomalies.missing_required),
            "duplicates": len(anomalies.duplicates),
            "outliers": len(anomalies.outliers),
        }
    }
    
    return stats, anomalies

def generate_report(data: dict, stats: Dict, anomalies: ServiceAnomalies):
    """Genera report dettagliato con anomalie e consigli"""
    
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    
    print("\n" + "=" * 70)
    print("🏥 REPORT AUDIT SERVIZI KB")
    print("=" * 70)
    
    print(f"\n📊 STATISTICHE GENERALI")
    print(f"   • Strutture totali: {stats['total_facilities']}")
    print(f"   • Servizi unici: {stats['total_unique_services']}")
    print(f"   • Anomalie rilevate: {sum(stats['anomaly_count'].values())}")
    
    # Dettaglio anomalie
    print(f"\n⚠️ ANOMALIE DETTAGLIATE")
    
    if anomalies.empty_services:
        print(f"\n   ❌ Strutture SENZA servizi ({len(anomalies.empty_services)}):")
        for a in anomalies.empty_services[:5]:  # Top 5
            print(f"      → {a['id']} ({a['tipologia']})")
    
    if anomalies.missing_required:
        print(f"\n   ❌ Servizi RICHIESTI mancanti ({len(anomalies.missing_required)}):")
        for a in anomalies.missing_required[:5]:
            print(f"      → {a['id']}: manca '{a['servizio_atteso']}'")
    
    if anomalies.duplicates:
        print(f"\n   ⚡ DUPLICATI in strutture ({len(anomalies.duplicates)}):")
        for a in anomalies.duplicates[:3]:
            print(f"      → {a['id']}: {a['duplicati']}")
    
    if anomalies.outliers:
        print(f"\n   🔍 Servizi RARI ({len(anomalies.outliers)}):")
        for a in anomalies.outliers[:5]:
            print(f"      → '{a['servizio']}' (offerto da {a['frequenza']} strutture)")
    
    # Salva report JSON
    report_json = {
        "timestamp": timestamp,
        "statistics": stats,
        "anomalies": {
            "empty_services": anomalies.empty_services,
            "missing_required": anomalies.missing_required,
            "duplicates": anomalies.duplicates,
            "outliers": anomalies.outliers,
        }
    }
    
    report_path = REPORT_DIR / f"audit_report_{timestamp}.json"
    with open(report_path, 'w', encoding='utf-8') as f:
        json.dump(report_json, f, indent=2, ensure_ascii=False)
    
    print(f"\n✅ Report salvato: {report_path}")
    
    return report_json

def export_normalized_kb(data: dict) -> str:
    """
    Esporta KB normalizzato (con servizi corretti).
    """
    facilities = data.get("facilities", [])
    
    for facility in facilities:
        services = facility.get("servizi_disponibili", [])
        if services:
            normalized = [normalize_service(s) for s in services]
            # Rimuovi duplicati mantenendo l'ordine
            facility["servizi_disponibili"] = list(dict.fromkeys(normalized))
    
    output_path = REPORT_DIR / "master_kb_normalized.json"
    with open(output_path, 'w', encoding='utf-8') as f:
        json.dump(data, f, indent=2, ensure_ascii=False)
    
    print(f"✅ KB normalizzato salvato: {output_path}")
    return str(output_path)

if __name__ == "__main__":
    try:
        print("🚀 Avvio audit servizi...\n")
        
        # Carica dati
        kb_data = load_kb(KB_FILE_PATH)
        
        # Esegui audit
        stats, anomalies = audit_services(kb_data)
        
        # Genera report
        report = generate_report(kb_data, stats, anomalies)
        
        # Opzionale: Esporta KB normalizzato
        print(f"\n📁 Esportazione KB normalizzato...")
        export_normalized_kb(kb_data)
        
        print("\n" + "=" * 70)
        print("✨ Audit completato!")
        print("=" * 70)
        
    except FileNotFoundError:
        print(f"❌ Errore: File {KB_FILE_PATH} non trovato.")
    except json.JSONDecodeError:
        print(f"❌ Errore: Il file {KB_FILE_PATH} non è un JSON valido.")
    except Exception as e:
        print(f"❌ Errore inaspettato: {e}")
